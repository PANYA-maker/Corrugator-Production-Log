⚙️ ตั้งค่า Google Apps Script Web App URL

วาง URL ที่ได้จากการ Deploy Google Apps Script Web App เพื่อเชื่อมต่อฐานข้อมูล Google Sheet

📖 วิธีติดตั้ง Google Apps Script ใน Google Sheet

  1. เปิด **Google Sheet** ใหม่ที่ต้องการบันทึกข้อมูล
  2. ไปที่เมนู **ส่วนขยาย (Extensions)** > **Apps Script**
  3. ลบโค้ดเดิมออกทั้งหมด แล้วนำโค้ดด้านล่างนี้ไปวาง:
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var payload = JSON.parse(e.postData.contents);
    var rows = payload.rows;
    var headers = payload.headers || [];
    
    // สร้าง Header ถ้าเป็นชีทว่าง
    if (sheet.getLastRow() === 0) {
      var headerRow = ["Timestamp"];
      for (var h = 0; h < headers.length; h++) {
        headerRow.push(headers[h]);
      }
      headerRow.push("Speed จริง", "ผลต่าง Speed", "สาเหตุ / หมายเหตุ", "วันที่เดินจริง", "กะที่เดิน");
      sheet.appendRow(headerRow);
    }
    
    // บันทึกข้อมูลแต่ละแถว
    for (var i = 0; i < rows.length; i++) {
      var r = rows[i];
      var dataRow = [new Date()];
      
      // ดึงค่าตามคอลัมน์ไดนามิก
      for (var c = 0; c < r.customVals.length; c++) {
        dataRow.push(r.customVals[c]);
      }
      
      dataRow.push(
        r.actualSpeed,
        r.actualSpeed - r.planSpeed,
        r.cause === 'อื่นๆ (ระบุในหมายเหตุ)' ? r.customNotes : r.cause,
        r.actualDate,
        r.shift
      );
      
      sheet.appendRow(dataRow);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", count: rows.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
