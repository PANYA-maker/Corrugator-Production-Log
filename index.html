📖 วิธีติดตั้ง Google Apps Script ใน Google Sheet

  1. เปิด **Google Sheet** ที่ต้องการบันทึกข้อมูล
  2. ไปที่เมนู **ส่วนขยาย (Extensions)** > **Apps Script**
  3. ลบโค้ดเดิมออกทั้งหมด แล้วนำโค้ดด้านล่างนี้ไปวาง:
// ======================================================
// 1. รับข้อมูลบันทึกจากหน้า Log (POST)
// ======================================================
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var payload = JSON.parse(e.postData.contents);
    var rows = payload.rows;
    var headers = payload.headers || [];
    
    if (sheet.getLastRow() === 0) {
      var headerRow = ["Timestamp"];
      for (var h = 0; h < headers.length; h++) {
        headerRow.push(headers[h]);
      }
      headerRow.push("Speed จริง", "ผลต่าง Speed", "สาเหตุ / หมายเหตุ", "วันที่เดินจริง", "กะที่เดิน", "เครื่องจักร");
      sheet.appendRow(headerRow);
    }
    
    for (var i = 0; i < rows.length; i++) {
      var r = rows[i];
      var dataRow = [new Date()];
      for (var c = 0; c < r.customVals.length; c++) {
        dataRow.push(r.customVals[c]);
      }
      dataRow.push(
        r.actualSpeed,
        r.actualSpeed - r.planSpeed,
        r.cause === 'อื่นๆ (ระบุในหมายเหตุ)' ? r.customNotes : r.cause,
        r.actualDate,
        r.shift,
        r.machine || 'YUELI'
      );
      sheet.appendRow(dataRow);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", count: rows.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// ======================================================
// 2. ส่งข้อมูลให้หน้า Dashboard (GET / JSONP)
// ======================================================
function doGet(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = sheet.getDataRange().getDisplayValues();
    var rows = [];
    
    if (data.length > 1) {
      var headers = data[0];
      for (var i = 1; i < data.length; i++) {
        var row = data[i];
        var obj = {};
        for (var j = 0; j < headers.length; j++) {
          var cleanKey = String(headers[j]).trim();
          obj[cleanKey] = row[j];
        }
        rows.push(obj);
      }
    }
    
    var result = { status: "success", data: rows };
    var callback = e && e.parameter ? e.parameter.callback : null;
    
    if (callback) {
      return ContentService.createTextOutput(callback + '(' + JSON.stringify(result) + ')')
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    } else {
      return ContentService.createTextOutput(JSON.stringify(result))
        .setMimeType(ContentService.MimeType.JSON);
    }
  } catch (err) {
    var errResult = { status: "error", message: err.toString() };
    var callback = e && e.parameter ? e.parameter.callback : null;
    if (callback) {
      return ContentService.createTextOutput(callback + '(' + JSON.stringify(errResult) + ')')
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    } else {
      return ContentService.createTextOutput(JSON.stringify(errResult))
        .setMimeType(ContentService.MimeType.JSON);
    }
  }
}
